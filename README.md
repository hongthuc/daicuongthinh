        let currentBanner = 0;
        const totalBanners = 2;
        let bannerInterval = null;

        function showBanner(index) {
            currentBanner = (index + totalBanners) % totalBanners;
            for (let i = 0; i < totalBanners; i++) {
                const slide = document.getElementById('bannerSlide' + i);
                const dot = document.getElementById('bannerDot' + i);
                if (i === currentBanner) {
                    slide.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
                    slide.classList.add('opacity-100', 'z-10');
                    dot.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                } else {
                    slide.classList.remove('opacity-100', 'z-10');
                    slide.classList.add('opacity-0', 'z-0', 'pointer-events-none');
                    dot.className = 'w-2.5 h-2 rounded-full bg-white/50 transition-all';
                }
            }
        }

        function changeBanner(step) {
            showBanner(currentBanner + step);
            resetBannerTimer();

            showBanner(index);
